}

rule("떡잎.gg | 경기 시작")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Is Game In Progress == True;
		Global.tnStarted == False;
	}

	actions
	{
		Global.tnStarted = True;
		Log To Inspector(Custom String("TN,start,{0},{1},{2}", Round To Integer(Total Time Elapsed, To Nearest), Current Map, Current Game Mode));
	}
}

rule("떡잎.gg | 영웅 선택과 교체")
{
	event
	{
		Ongoing - Each Player;
		All;
		All;
	}

	conditions
	{
		Has Spawned(Event Player) == True;
		Is Dummy Bot(Event Player) == False;
		Hero Of(Event Player) != Event Player.tnHero;
	}

	actions
	{
		Event Player.tnHero = Hero Of(Event Player);
		Log To Inspector(Custom String("TN,hero,{0},{1},{2}", Round To Integer(Total Time Elapsed, To Nearest), Event Player, Custom String("{0},{1},{2}", (Team Of(Event Player) == Team 1) ? 1 : 2, Array Contains(All Tank Heroes, Hero Of(Event Player)) ? 1 : (Array Contains(All Support Heroes, Hero Of(Event Player)) ? 3 : 2), Hero Of(Event Player))));
	}
}

rule("떡잎.gg | 처치")
{
	event
	{
		Player Died;
		All;
		All;
	}

	conditions
	{
		Global.tnStarted == True;
	}

	actions
	{
		Log To Inspector(Custom String("TN,kill,{0},{1},{2}", Round To Integer(Total Time Elapsed, To Nearest), Attacker, Custom String("{0},{1},{2}", Hero Of(Attacker), Victim, Custom String("{0},{1},{2}", Hero Of(Victim), (Event Ability == Button(Ultimate)) ? 5 : ((Event Ability == Button(Melee)) ? 6 : ((Event Ability == Button(Ability 1)) ? 3 : ((Event Ability == Button(Ability 2)) ? 4 : ((Event Ability == Button(Secondary Fire)) ? 2 : ((Event Ability == Button(Primary Fire)) ? 1 : 0))))), Custom String("{0},{1},{2}", (Team Of(Attacker) == Team 1) ? 1 : 2, (Team Of(Victim) == Team 1) ? 1 : 2, Custom String("{0},{1}", Count Of(Players Within Radius(Position Of(Attacker), 15, Team Of(Attacker), Off)), Count Of(Players Within Radius(Position Of(Attacker), 15, Opposite Team Of(Team Of(Attacker)), Off))))))));
	}
}

rule("떡잎.gg | 경기 끝 - 점수")
{
	event
	{
		Ongoing - Global;
	}

	conditions
	{
		Global.tnStarted == True;
		Is Match Complete == True;
	}

	actions
	{
		Wait(1, Ignore Condition);
		Log To Inspector(Custom String("TN,end,{0},{1},{2}", Round To Integer(Total Time Elapsed, To Nearest), Team Score(Team 1), Team Score(Team 2)));
		Global.tnStarted = False;
	}
}
